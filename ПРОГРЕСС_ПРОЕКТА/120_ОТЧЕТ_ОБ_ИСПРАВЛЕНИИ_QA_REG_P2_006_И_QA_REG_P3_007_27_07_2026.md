
```text

```

После исправления:

```text

```

Аудит синтаксически ищет прямой `ast.Return`, после которого в том же
последовательном AST-блоке остаются statements. Он намеренно не является
полным CFG-анализом и не ищет недостижимость после `raise`, `break`,
`continue`, `sys.exit()` или функций, которые фактически всегда завершают
выполнение.

### 11.8. Повторить AST-контроль

Из `СИСТЕМА_MVP/backend`:

```powershell
@'
import ast
from pathlib import Path

files = sorted(p for p in Path('.').rglob('test*.py') if p.is_file())
findings = []

def scan(node, path, scope=()):
    if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
        scope = (*scope, node.name)
    for field, value in ast.iter_fields(node):
        if isinstance(value, list):
            if value and all(isinstance(item, ast.stmt) for item in value):
                for index, statement in enumerate(value[:-1]):
                    if isinstance(statement, ast.Return):
                        trailing = value[index + 1:]
                        findings.append((
                            path.as_posix(),
                            '.'.join(scope) or '<module>',
                            field,
                            statement.lineno,
                            [(item.lineno, type(item).__name__) for item in trailing],
                        ))
            for item in value:
                if isinstance(item, ast.AST):
                    scan(item, path, scope)
        elif isinstance(value, ast.AST):
            scan(value, path, scope)

for path in files:
    tree = ast.parse(
        path.read_text(encoding='utf-8-sig'),
        filename=str(path),
    )
    scan(tree, path)

print(f'FILES={len(files)}')
print(f'RETURN_BLOCK_FINDINGS={len(findings)}')
for path, scope, field, line, trailing in findings:
    tail = ', '.join(
        f'{kind}@{lineno}'
        for lineno, kind in trailing
    )
    print(
        f'{path}:{line}: {scope}.{field} -> '
        f'{len(trailing)} trailing statement(s): {tail}'
    )

raise SystemExit(1 if findings else 0)
'@ | ..\.venv\Scripts\python.exe -
```
